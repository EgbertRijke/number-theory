% Blackboard Ally / Canvas compatibility master.
%
% VERIFIED ALLY PROFILE
% ---------------------
% This profile received an Ally accessibility score of 100 percent for
% number-theory-volume-1-ally.pdf in JHU Canvas on 2026-09-01.  The score
% depends on the complete set of compatibility measures below.  They are
% intentionally confined to the Ally edition: the canonical edition retains
% PDF/UA-2, PDF/A-4f, and structured MathML.
%
% 1. PDF and mathematics (setup/document-metadata.tex)
%
%    When \NTAllyCompatibilityBuild is defined, the active metadata is:
%
%      \DocumentMetadata{
%        lang          = en-US,
%        pdfversion    = 1.7,
%        pdfstandard   = ua-1,
%        tagging       = on,
%        tagging-setup = {
%          math/setup={},
%          math/alt/use,
%          extra-modules={verbatim-mo}
%        }
%      }
%
%    PDF/A-4f and math/setup=mathml-SE are deliberately absent.  Ally does
%    not currently understand the PDF/UA-2 MathML structure and otherwise
%    reports formulas as undescribed images.  math/alt/use supplies the
%    formula /Alt entries expected by Ally's PDF/UA-1-oriented checks.
%
% 2. Document title (number-theory.tex, volume-N.tex, setup/references.tex)
%
%    Every master defines \title and \author before loading the preamble.
%    hyperref is loaded with pdfusetitle, and pdfdisplaydoctitle is true.
%    In the PDF 1.7 / PDF/UA-1 profile this exposes the title in the legacy
%    metadata location recognized by Ally.
%
% 3. Color contrast (setup/colors.tex)
%
%    CornflowerBlue is rebound locally to CornflowerBlueHighContrast
%    (HTML 007ACD).  This also changes hyperref's allcolors value and every
%    other use of CornflowerBlue as a text color in the Ally edition.
%
% 4. Headerless visual grids (at the individual tabular)
%
%    A tabular used only to lay out a figure or other visual grid, and which
%    therefore has no genuine row or column headers, must disable structural
%    table tagging locally:
%
%      \begin{figure}
%        \tagpdfsetup{table/tagging=false}
%        ...
%        \begin{tabular}{...}
%          ...
%        \end{tabular}
%        \caption{...}
%      \end{figure}
%
%    Do not manufacture meaningless header cells.  Although
%    table/tagging=presentation is semantically correct, Ally currently
%    ignores its ARIA presentation role and still reports a table without
%    headers.  table/tagging=false removes the Table/TR/TD hierarchy while
%    preserving the cell contents and their supplied ActualText.
%

%

%   make ally volume 1
%   make ally volumes
%

% before \documentclass is processed by number-theory.tex.
