\begin{proposition}
  Consider three integers $a$, $b$, and $c$ satisfying the equation
  \begin{equation*}
    a^n+b^n=c^n,
  \end{equation*}
  where $n\geq 1$. If one of the following three properties holds, then so do the other two:
  \begin{enumerate}
  \item $\gcd(a,b)=1$.
  \item $\gcd(a,c)=1$.
  \item $\gcd(b,c)=1$.
  \end{enumerate}
\end{proposition}

\begin{lemma}\label{lemma:binary-fermat-quotient}
  Consider a primitive triple of positive integers $a$, $b$, and $c$ and an odd prime $p$ such that $a^p+b^p=c^p$. Define the \emph{binary Fermat quotient}
  \begin{equation*}
    q_p(a,b):=\sum_{k=0}^{p-1} (-1)^k a^{p-1-k}b^k,
  \end{equation*}
  so that $a^p+b^p=(a+b)\cdot q_p(a,b)$. Then we have
  \begin{equation*}
    \gcd(a+b,q_p(a,b))=
    \begin{cases}
      1 & \text{if $p\nmid a+b$,}\\
      p & \text{if $p\mid a+b$.}
    \end{cases}
  \end{equation*}
\end{lemma}

\begin{proof}
  Consider the modulus $m:=a+b$ so that $a\equiv -b\pmod{m}$. Then we have
  \begin{equation*}
    q_p(a,b)=\sum_{k=0}^{p-1}(-1)^ka^{p-1-k}b^k \equiv \sum_{k=0}^{p-1} (-1)^ka^{p-1-k}(-a)^k= pa^{p-1} \pmod{m}.
  \end{equation*}
  It follows that $\gcd(a+b,q_p(a,b))$ divides $\gcd(a+b,pa^{p-1})$. However, since $a$ is relatively prime to $a+b$, we find that the latter greatest common divisor divides $p$.
\end{proof}

\begin{theorem}
  Consider a  natural number $n$, and write $q=\prod_{p-1\mid n-1}p$ for the product of all primes $p$ such that $p-1\mid n-1$. Then
  \begin{equation*}
    a^n\equiv a \pmod{q}
  \end{equation*}
  for every integer $a$. Consequently, if $x$, $y$, and $z$ are integers such that $x^n+y^n=z^n$, then we have
  \begin{equation*}
    x+y\equiv z\pmod{q}.
  \end{equation*}
\end{theorem}
