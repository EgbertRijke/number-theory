\usepackage{totcount}

\newtotcounter{exercisecounter}
\AtEndDocument{%
  \typeout{Total number of exercises: \number\value{exercisecounter}}%
}

\resetenumext[1]{chapter}

\NewDocumentCommand{\exercises}{}{%
  \par\clearpage%
  \phantomsection%
  \section*{Exercises}%
  \addcontentsline{toc}{section}{Exercises}%
  \markright{Exercises}%
}

\NewDocumentCommand{\exercise}{}{%
  \stepcounter{exercisecounter}%
  \item
}

\NewDocumentEnvironment{exenum}{}{%
  \begin{enumext}[
    resume,
    label=\thechapter.\arabic*,
    ref=\thechapter.\arabic*,
    widest=99,
    labelsep=.6em,
    noitemsep]
}{%
  \end{enumext}
}

\NewDocumentEnvironment{subexenum}{}{%
  \begin{enumext}[
    label=(\alph*),
    ref=\thechapter.\arabic{enumXi}(\alph*),
    noitemsep]
}{%
  \end{enumext}
}

\NewDocumentEnvironment{starter}{}{%
  \italicsubsection{Starter Exercises}%
  \begin{exenum}
}{%
  \end{exenum}
}

  \italicsubsection{Foundational Exercises}%

  \end{exenum}

  \end{exenum}
