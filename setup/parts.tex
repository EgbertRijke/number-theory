\AtBeginDocument{%
  \renewcommand{\partname}{Book}%
}

\newlength{\BookDecorationSep}
\setlength{\BookDecorationSep}{3\Hugebaselineskip}

\ExplSyntaxOn

\tl_new:N \g__nt_book_decoration_tl

\NewDocumentCommand \BookDecoration { O{} +m }
  {
    \tl_gset:Nn \g__nt_book_decoration_tl
      {
        \begin{tikzpicture}[artifact,#1]
          #2
        \end{tikzpicture}
      }
  }

\NewDocumentCommand \NoBookDecoration { }
  {
    \tl_gclear:N \g__nt_book_decoration_tl
  }

\NewDocumentCommand \PrintBookDecoration { }
  {
    \tl_if_empty:NF \g__nt_book_decoration_tl
      {
        \par
        \group_begin:
          \normalfont\normalsize
          \centering
          \skip_zero:N \parskip
          \noindent
          \tl_use:N \g__nt_book_decoration_tl
          \par
        \group_end:
        \nobreak
        \vspace{\BookDecorationSep}
        \tl_gclear:N \g__nt_book_decoration_tl
      }
  }

\ExplSyntaxOff

\makeatletter

\newlength{\BookNumberTitleSep}
\setlength{\BookNumberTitleSep}{\Hugebaselineskip}

\EditInstance{headformat}{part}{
  number-title-sep = 1.5\BookNumberTitleSep
}

\EditInstance{heading}{part}{
  start-code   =
    {
      \if@openright
        \cleardoublepage
      \else
        \clearpage
      \fi
      \thispagestyle{empty}
      \if@twocolumn
        \onecolumn
        \@tempswatrue
      \else
        \@tempswafalse
      \fi
      \null\vfil
      \PrintBookDecoration
    },
  prefix-decls = \partdisplayfont\mdseries\lsstyle\scshape,
  number-decls = \partdisplayfont\mdseries\lsstyle\scshape,
  title-decls  = \partdisplayfont\mdseries\upshape\Huge\lsstyle\scshape
}

\makeatother
