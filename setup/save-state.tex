% Keep each explicit font build in its own auxiliary namespace.  The Makefile
% supplies either an empty suffix or a validated value such as -alegreya.
\directlua{
  local suffix = os.getenv("NT_VOLUME_SUFFIX") or ""
  local valid = suffix == "" or suffix:sub(1, 1) == "-"
  for i = 1, suffix:len() do
    local byte = suffix:byte(i)
    valid = valid and (
      byte == 45
      or (byte >= 48 and byte <= 57)
      or (byte >= 65 and byte <= 90)
      or (byte >= 97 and byte <= 122)
    )
  end
  if valid == false then
    tex.error("Invalid NT_VOLUME_SUFFIX", {suffix})
  end
  token.set_macro("NTVolumeSuffix", suffix, "global")
}

\NewExpandableDocumentCommand{\VolumeAuxPath}{m}{%
  build/number-theory-volume-#1\NTVolumeSuffix
}

\NewExpandableDocumentCommand{\VolumePdfPath}{m}{%
  number-theory-volume-#1\NTVolumeSuffix.pdf
}

% Expand the suffix before zref stores the external auxiliary and PDF names.
\NewDocumentCommand{\ExternalVolume}{m}{%
  \begingroup
  \edef\next{%
    \endgroup
    \noexpand\zexternaldocument
      {\VolumeAuxPath{#1}}%
      [\VolumePdfPath{#1}]%
  }%
  \next
}

\zref@newprop{volchapter}[0]{\number\value{chapter}}
\zref@newprop{volnextpage}[1]{\number\value{page}}

\NewDocumentCommand{\SaveVolumeChapterState}{m}{%
  \cleardoublepage
  \zref@wrapper@immediate{%
    \zref@labelbyprops
      {volume-chapter:#1}
      {volchapter}%
  }%
}

\NewDocumentCommand{\SaveVolumePageState}{m}{%
  \cleardoublepage
  \zref@wrapper@immediate{%
    \zref@labelbyprops
      {volume-page:#1}
      {volnextpage}%
  }%
}

\NewDocumentCommand{\ContinueFromVolume}{m}{%
  \zref@ifrefundefined{volume-chapter:#1}{%
    \PackageError{number-theory}
      {Missing chapter state for #1}
      {Build the preceding volume first.}%
  }{}%
  \zref@ifrefundefined{volume-page:#1}{%
    \PackageError{number-theory}
      {Missing page state for #1}
      {Build the preceding volume first.}%
  }{}%
  \setcounter{chapter}{%
    \zref@extractdefault
      {volume-chapter:#1}
      {volchapter}
      {0}%
  }%
  \setcounter{page}{%
    \zref@extractdefault
      {volume-page:#1}
      {volnextpage}
      {1}%
  }%
}

\makeatother
