\usepackage{amsmath}
\usepackage{amsthm}

\newtheoremstyle{nttheorem}
  {}{}                            % Default vertical spacing
  {\itshape}                      % Italic body
  {}                              % No indentation
  {\normalfont\scshape}            % Small-caps heading
  {.}
  {5pt plus 1pt minus 1pt}         % Default heading separation
  {}

\newtheoremstyle{ntdefinition}
  {}{}                            % Default vertical spacing
  {\normalfont}                   % Upright body
  {}
  {\normalfont\scshape}
  {.}
  {5pt plus 1pt minus 1pt}
  {}

\newtheoremstyle{ntremark}
  {}
  {}
  {\normalfont}
  {}
  {\normalfont\scshape}
  {.}
  {5pt plus 1pt minus 1pt}
  {}

\theoremstyle{nttheorem}
\newtheorem{theorem}{Theorem}[section]
\newtheorem{lemma}[theorem]{Lemma}
\newtheorem{proposition}[theorem]{Proposition}
\newtheorem{corollary}[theorem]{Corollary}
\newtheorem{conjecture}{Conjecture}
\newtheorem{problem}[conjecture]{Problem}

\theoremstyle{ntdefinition}
\newtheorem{definition}[theorem]{Definition}
\newtheorem{example}[theorem]{Example}

\theoremstyle{ntremark}
\newtheorem{remark}[theorem]{Remark}
