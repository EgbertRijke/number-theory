  {

      }
  }

  {
    \int_if_odd:nTF { \ReadonlyShipoutCounter }

          { br }
          {
            \__coffee_message:n { \raggedleft }
            \hspace{3mm}
            \__coffee_qr:
          }
      }
      {
        \__coffee_put:nnn
          { \g__coffee_left_inset_dim }
          { bl }
          {
            \__coffee_qr:
            \hspace{3mm}
            \__coffee_message:n { \raggedright }
          }
      }
  }
