\begin{exercises}
  \exercise Solve the following linear congruences:
  \begin{subexenum}
  \item $3x\equiv 7\pmod{11}$,
  \item $12x\equiv 18\pmod{30}$,
  \item $7x\equiv 14\pmod{21}$.
  \end{subexenum}
  Describe the congruence classes of all solutions, if multiple solutions are possible.
  \exercise Find the smallest positive integer $x$ such that
  \begin{align*}
    x & \equiv 2 \pmod{3} \\
    x & \equiv 5 \pmod{7} \\
    x & \equiv 6 \pmod{11}.
  \end{align*}
  \exercise Determine the residue class of
  \begin{equation*}
    2^{100}+3^{100}
  \end{equation*}
  modulo $13$.
  \exercise Determine the residue class of
  \begin{equation*}
    7^{4^{50}}
  \end{equation*}
  modulo $13$.
  \exercise
  \begin{subexenum}
  \item Find the least primitive root $g$ modulo $19$, and give a table of the congruence classes of its powers.
  \item Find all primitive roots modulo $19$.
  \item For each $k\mid\phi(19)$, find all elements of order $k$ modulo $19$.
  \item For each integer $1,\ldots,18$, find its index base $g$ modulo $19$.
  \item Solve the linear congruence
    \begin{equation*}
      7x\equiv 8\pmod{19}
    \end{equation*}
    using indices.
  \end{subexenum}
\end{exercises}
