\directlua{
  fonts.handlers.otf.addfeature{
    name = "rijkeliga",
    type = "chainsubstitution",
    lookups = {
      {
        type = "ligature",
        data = {
          ["ĳ"] = { "i", "j" },
        }
      }
    },
    data = {
      rules = {
        {
          before  = { { "R" } },
          current = { { "i" }, { "j" } },
          after   = { { "k" }, { "e" } },
          lookups = { 1 }
        }
      }
    }
  }
}
