\begin{exercises}
  \exercise (10 points) Recall that the Fibonacci sequence $F$ is given by
  \begin{equation*}
    F_0 := 0, \quad F_1 := 1, \qquad\text{and}\qquad F_{n+2}=F_{n+1}+F_n.
  \end{equation*}
  Prove by induction that
  \begin{equation*}
    1 + \sum_{k=0}^{n} F_k = F_{n+2}.
  \end{equation*}
  \exercise (10 points) Use Euclid's algorithm to compute the following greatest common divisors, and express them as linear combinations via B\'ezout's identity:
  \begin{subexenum}
  \item $\gcd(396,312)$
  \item $\gcd(765,525)$
  \item $\gcd(408,169)$
  \end{subexenum}
  \exercise (10 points) Prove that for any natural number $n>0$ there is a largest power of $2$ dividing $n$, i.e., there is a natural number $k$ such that for every natural number $l$ we have
  \begin{equation*}
    2^l \mid n \Leftrightarrow l \leq k.
  \end{equation*}
  Make sure to be explicit about how you use the well-ordering theorem, strong induction, or ordinary induction.
  \exercise (10 points) Show that for any natural number $n>0$ there is a unique pair of numbers $(k,m)$ such that
  \begin{equation*}
    n = 2^k(2m+1).
  \end{equation*}
  This is called the $2$-adic representation of $n$.
\end{exercises}
