\begin{exercises}
  \exercise Solve the following linear congruences:
  \begin{subexenum}
  \item $5x\equiv 8\pmod{13}$,
  \item $14x\equiv 7\pmod{19}$.
  \end{subexenum}
  Describe the congruence classes of all solutions, if multiple solutions are possible.
  \exercise Find the smallest positive integer $x$ such that
  \begin{align*}
    x & \equiv 1 \pmod{4} \\
    x & \equiv 3 \pmod{5} \\
    x & \equiv 8 \pmod{13}.
  \end{align*}
  \exercise Determine the residue class of
  \begin{equation*}
    4^{100}+5^{100}
  \end{equation*}
  modulo $7$.
  \exercise Determine the residue class of
  \begin{equation*}
    7^{3^{33}}
  \end{equation*}
  modulo $11$.
  \exercise
  \begin{subexenum}
  \item Find the least primitive root $g$ modulo $17$, and give a table of the congruence classes of its powers.
  \item Find all primitive roots modulo $17$.
  \item For each $k\mid\phi(17)$, find all elements of order $k$ modulo $17$.
  \item For each integer $1,\ldots,16$, find its index base $g$ modulo $17$.
  \item Solve the linear congruence
    \begin{equation*}
      11x\equiv 8\pmod{17}
    \end{equation*}
    using indices.
  \end{subexenum}
\end{exercises}
