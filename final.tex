\begin{exercises}
  \exercise Prove that
  \begin{equation*}
    \prod_{k=2}^n\left(1-\frac{1}{k}\right)=\frac{1}{n}.
  \end{equation*}
  \exercise Compute $\gcd(4321,9876)$.
  \exercise Describe all the integer solutions to the equation $105x+121y=1$.
  \exercise Use Legendre's Formula or Kummer's Theorem to show that
  \begin{equation*}
    \binom{35}{7}\,\Bigm|\,\binom{35}{14}.
  \end{equation*}
  \exercise Find all incongruent solutions of the linear congruence $21x\equiv 14\pmod{91}$.
  \exercise Solve $x^{38}\equiv 3\pmod{13}$.
  \exercise Find two different positive integers $n$ for which $\phi(n)=16$.
  \exercise How many even positive integers $1< n\leq 100$ are there, which are not divisible by $3$, $5$, and $7$?
  \exercise Determine whether the congruence $x^2+14x-35\equiv 0\pmod{57}$ has a solution.
  \exercise Compute the Legendre symbols
  \begin{equation*}
    \legendre{199}{919}\qquad\legendre{919}{991}\qquad\text{and}\qquad \legendre{991}{199}.
  \end{equation*}
\end{exercises}
