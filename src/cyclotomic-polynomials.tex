\section{The Primitive Factors of \texorpdfstring{$x^n-1$}{xⁿ-1}}

Consider a prime factor $p$ of $a^n-1$. Since $p\mid a^n-1$, it follows that $a^n\equiv 1\pmod{p}$. By the \zcrefnamed{theorem:order-theorem} it follows that $\ord_p(a)\mid n$. By \zcrefnamed{theorem:fermats-little-theorem} it follows that the order of $a$ modulo $p$ is itself a divisor of $p-1$.

\begin{definition}
  The $n$\supth{} \define{cyclotomic polynomial} $\Phi_n$ is defined by
  \begin{equation*}
    \Phi_n(x) := \prod_{\substack{1\leq k \leq n\\[2pt]\gcd(k,n)=1}}(x-e^{2\pi ik/n}).
  \end{equation*}
\end{definition}


\begin{example}
  Another way of expressing the defining identity of the cyclotomic polynomial $\Phi_{pq}(x)$ is
  \begin{equation*}
    x^{pq}-1=\frac{(x^p-1)(x^q-1)}{x-1}\Phi_{pq}(x).
  \end{equation*}
  In other words, $\Phi_{pq}(x)$ can be expressed as the following fraction of polynomials:
  \begin{equation*}
    \frac{(x^{pq}-1)(x-1)}{(x^p-1)(x^q-1)}=\Phi_{pq}(x).
  \end{equation*}
  The idea is now to write $(x^{pq}-1)(x-1)$ as a combination of polynomials of the form $(x^{up}-1)(x^{vq}-1)$.
  Note first that $p$ is invertible modulo $q$ and $q$ is invertible modulo $p$. Thus, there are integers $0\leq s<q$ and $0\leq t<p$ such that $sp\equiv 1\pmod{q}$ and $tq\equiv 1\pmod{p}$. Combining these congruences, we obtain
  \begin{equation*}
    sp+tq\equiv 1\pmod{pq}.
  \end{equation*}
  Moreover, we have $0\leq sp+tq<pq+pq$, so that in fact the identity
  \begin{equation*}
    sp+tq=pq+1
  \end{equation*}
  holds. With this choice of $s$ and $t$, we calculate
  \begin{align*}
    (x^{pq}-1)(x-1)-(x^{sp}-1)(x^{tq}-1)
    & =x^{pq+1}-x^{pq}-x+1-x^{sp+tq}+x^{sp}+x^{tq}-1 \\
    & = -x(x^{pq-1}-x^{sp-1}-x^{tq-1}+1) \\
    & = -x(x^{sp-1}-1)(x^{tq-1}-1) \\
    & = -x(x^{(q-s)p}-1)(x^{(p-t)q}-1).
  \end{align*}
  This shows that
  \begin{equation*}
    (x^{pq}-1)(x-1)=(x^{sp}-1)(x^{tq}-1)-x(x^{(q-s)p}-1)(x^{(p-t)q}-1).
  \end{equation*}
  This shows that
  \begin{equation*}
    \Phi_{pq}(x)=\frac{(x^{sp}-1)(x^{tq}-1)}{(x^p-1)(x^q-1)}-x\frac{(x^{(q-s)p}-1)(x^{(p-t)q}-1)}{(x^p-1)(x^q-1)}.
  \end{equation*}
  In other words, we find that
  \begin{equation*}
    \Phi_{pq}(x)=\left(\sum_{i=0}^{s-1}x^{ip}\right)\left(\sum_{j=0}^{t-1}x^{jq}\right)-x\left(\sum_{i=0}^{q-s-1}x^{ip}\right)\left(\sum_{j=0}^{p-t-1}x^{jq}\right).
  \end{equation*}
\end{example}


\begin{theorem}
  For any natural number $n$ we have the identity
  \begin{equation*}
    \Phi_n(x)=\prod_{\substack{d\mid n\\[2pt]d\neq n}}\Phi_d(x).
  \end{equation*}
\end{theorem}

\begin{theorem}
  The cyclotomic polynomials $\Phi_n(x)$ have integer coefficients
\end{theorem}

In the following theorem we will establish that the coefficients of the cyclotomic polynomials are palindromic, meaning that they read the same forward and back. 

\begin{theorem}
  The cyclotomic polynomials $\Phi_n(x)$ are \define{reciprocal} in the sense that
  \begin{equation*}
    \Phi_n\left(\tfrac{1}{x}\right)x^{\phi(n)} = \Phi_n(x).
  \end{equation*}
\end{theorem}

\begin{theorem}
  Consider a cyclotomic polynomial $\Phi_n(x)$ and a prime $p$. Then we have
  \begin{equation*}
    \Phi_n(x)\equiv 0\pmod{p}\text{ is solvable}\qquad\Leftrightarrow\qquad n\mid p-1.
  \end{equation*}
  Furthermore, if $\Phi_n(x)\equiv 0\pmod{p}$ is solvable, then $\Phi_n(x)$ has exactly $\phi(n)$ roots modulo $p$.
\end{theorem}

\begin{definition}
  We define
  \begin{equation*}
    \Phi_n(x,y):= y^{\totient(n)}\Phi_n\left(\tfrac{x}{y}\right).
  \end{equation*}
\end{definition}

\begin{proposition}
  For any $x$, $y$, and $n$, we have:
  \begin{equation*}
    x^n-y^n=\prod_{d\mid n}\Phi_d(x,y).
  \end{equation*}
\end{proposition}

\begin{lemma}
  Consider integers $a$, $b$, and $c$, and a prime $p$ such that $p\nmid ab$ and $c\equiv a/b\pmod{p}$. If $\ord_p(c)=k$, then
  \begin{equation*}
    \Phi_k(c)\equiv 0\pmod{p} \Leftrightarrow p\mid \Phi_k(a,b)
  \end{equation*}
\end{lemma}

\begin{lemma}
  If $p\nmid m$, then $\Phi_m(x^p)=\Phi_{mp}(x)\Phi_m(x)$.
\end{lemma}

\begin{lemma}
  Consider a prime $p$, and let $n=p^\alpha m$, where $\alpha\geq 1$ and $p\nmid m$. Then we have
  \begin{equation*}
    \Phi_n(x)\equiv \Phi_m(x)^{\phi(p^\alpha)}\pmod{p}.
  \end{equation*}
  Consequently, if $p\mid\Phi_n(a,b)$ then $p\mid \Phi_m(a,b)$.
\end{lemma}

\begin{proposition}
  Consider a prime $p\nmid n$. Then 
  \begin{equation*}
    \ord_p(a)= n \Leftrightarrow p\mid \Phi_n(a)
  \end{equation*}
\end{proposition}

\section{The M\"obius Inversion Formula for the Cyclotomic Polynomials}

\begin{theorem}\label{theorem:mobius-inversion-cyclotomic-polynomials}
  For any $n$, we have the identity
  \begin{equation*}
    \Phi_n(x)=\prod_{d\mid n}(x^d-1)^{\mu(n/d)}
  \end{equation*}
\end{theorem}

\begin{example}
  \zcref{theorem:mobius-inversion-cyclotomic-polynomials} can be used effectively to compute the coefficients of a cyclotomic polynomial. For example, in order to compute $\Phi_{12}(x)$, we note that the values $\mu(12/d)$ of the M\"obius function are as follows:
  \begin{equation*}
    \mu(12/1)=0,~\mu(12/2)=1,~\mu(12/3)=0,~\mu(12/4)=-1,~\mu(12/6)=-1,~\mu(12/12)=1.
  \end{equation*}
  This shows that
  \begin{equation*}
    \Phi_{12}(x)=(x^2-1)(x^4-1)^{-1}(x^6-1)^{-1}(x^{12}-1)=x^4-x^2+1.
  \end{equation*}
\end{example}

\section{Irreducibility of the Cyclotomic Polynomials}

\begin{definition}
  We say that a polynomial $f(x)$ is \define{square-free} if $r(x)^2\mid x^n-1$ implies that $r(x)$ is constant, for every polynomial $r(x)$. Furthermore, we say that $f(x)$ is \define{square-free modulo $p$} if $r(x)^2\mid x^n-1 \pmod{p}$ implies that $r(x)$ is constant modulo $p$, for every polynomial $r(x)$.
\end{definition}

The polynomial $x^n-1$ is square-free in the ordinary sense, since it does not have repeated roots. In the following lemma we will prove the stronger property that $x^n-1$ is square-free modulo primes $p$ not dividing $n$.

\begin{lemma}
  The polynomial $x^n-1$ is square-free modulo $p$ for every prime $p\nmid n$.
\end{lemma}

\begin{proof}
  It suffices that there cannot be an irreducible polynomial $r(x)$ such that
  \begin{equation*}
    x^n-1=r(x)^2s(x)\pmod{p}.
  \end{equation*}
  By taking derivatives, we see that
  \begin{equation*}
    nx^{n-1}=2r(x)r'(x)s(x)+r(x)^2s(x)\pmod{p}.
  \end{equation*}
  In particular, we see that $r(x)\mid nx^{n-1}$ in $\mathbb{F}_p[x]$. By the assumption that $p\nmid n$ it follows that $n$ is invertible, and hence we obtain
  \begin{equation*}
    r(x)\mid x^{n-1}.
  \end{equation*}
  Since $r(x)$ is assumed to be irreducible, it follows that $r(x)\mid x$. That is, $x\mid r(x)$. This leads to a contradiction, however, since
  \begin{equation*}
    x\nmid x^n-1.
  \end{equation*}
\end{proof}

\begin{theorem}
  The $n$\supth cyclotomic polynomial $\Phi_n(x)$ is irreducible for every $n\geq 1$.
\end{theorem}

\begin{proof}
  Let $\zeta=e^{2\pi i/n}$. We claim that if $f(x)$ is an irreducible polynomial such that $f(\zeta)=0$, then $f(\zeta^a)=0$ for every integer $a$ coprime to $n$. To prove this claim, it suffices to show that $f(\zeta^p)=0$ for every prime $p\nmid n$. 

  We observe first that $\zeta^p$ is a primitive root of unity, since $(\zeta^p)^m=1$ implies $n\mid pm$, which implies $n\mid m$ since $p\nmid n$. Now, since we have assumed that $f(\zeta)=0$ it follows that
  \begin{equation*}
    \Phi_n(x)=f(x)g(x).
  \end{equation*}
Therefore, if $f(\zeta^p)\neq 0$, it follows that $g(\zeta^p)=0$. This shows that $\zeta$ is a root of the polynomial $g(x^p)$. Since $f(x)$ is an irreducible polynomial with root $\zeta$, it follows that $f(x)\mid g(x^p)$.

  As a consequence of the fact that $\zeta^a$ is a root of every irreducible polynomial $f(x)$ such that $f(\zeta)=0$, for every integer $a$ relatively prime to $n$, every such polynomial is of degree $\varphi(n)$. It follows that the polynomial $\Phi_n(x)$ is of minimal degree such that $\Phi_n(\zeta)=0$, and therefore must be irreducible.
\end{proof}

\section{Zsigmondy's Theorem}

\begin{definition}
  Consider a sequence $(a_n)$ of integers. A prime $p$ is said to be a \define{primitive prime divisor} of $a_n$ if the rank of apparition of $p$ satisfies $\rho(p)=n$.
\end{definition}

\begin{proposition}\label{proposition:zsigmondy}
  Consider two relatively prime integers $a$ and $b$, and a prime $p$ such that $p\mid \Phi_n(a,b)$, where $n\geq 1$. Then there are two mutually exclusive possibilities:
  \begin{enumerate}
  \item Either $p\equiv 1\pmod{n}$, in which case $p$ is a primitive prime divisor of $a^n-b^n$, or
  \item $p\mid n$, in which case $p$ is the largest prime factor of $n$, and provided that $n>2$ we have furthermore $p^2\nmid \Phi_n(a,b)$.
  \end{enumerate}
\end{proposition}

\begin{proof}
  Since $a^n-b^n=\prod_{d\mid n}\Phi_d(a,b)$, it follows that $a^n\equiv b^n\pmod{p}$. Furthermore $a$ and $b$ are assumed to be relatively prime, the congruence $a^n\equiv b^n\pmod{p}$ implies that neither $a$ nor $b$ is divisible by $p$. We find that
  \begin{equation*}
    (a/b)^n\equiv 1\pmod{p},
  \end{equation*}
  and hence $k:=\ord_p(a/b)$ is a divisor of $n$ by \zcrefnamed{theorem:order-theorem}. There are two cases to consider: Either $k=n$ or $k$ is a proper divisor of $n$.

  If $k=n$, then it follows by \zcrefnamed{theorem:fermats-little-theorem} that $n\mid p-1$, which is another way of saying that $p\equiv 1\pmod{n}$. Since $k\mid m$ for any $m$ such that $p\mid a^m-b^m$, it follows that $p$ is a primitive divisor of $a^n-b^n$.
  
  For the second case, suppose that $k$ is a proper divisor of $n$. Since $p\mid \Phi_k(a,b)$ and $p\mid \Phi_n(a,b)$ both hold, it follows that the polynomial congruence
  \begin{equation*}
    x^n-1\pmod{p}
  \end{equation*}
  has roots of multiplicity $>1$. It follows that
  \begin{equation*}
    \gcd(x^n-1,nx^{n-1})\neq 1,
  \end{equation*}
  from which we conclude that $p\mid n$.

  To show that $p$ is the largest prime factor of $n$, write $n=p^\alpha m$, where $p\nmid m$. By the implication
  \begin{equation*}
    p\mid \Phi_n(a,b)\Rightarrow p\mid \Phi_m(a,b)
  \end{equation*}
  it follows that $p\mid \Phi_m(a,b)$. However, since $p\nmid m$ this brings us back to the first case with the exponent $m$, so we conclude that $m\mid p-1$. In particular, any prime divisor of $m$ is a prime divisor of $p-1$, which implies that $p$ must have been the largest prime divisor of $n$.

  Finally, we have to show that $\nu_p(\Phi_n(a,b))=1$. We consider two cases: Either $p=2$ or $p$ is odd. If $p=2$, write $n=2^\alpha m$. As before, it follows that $m\mid p-1$, so we have $m=1$ and $n=2^\alpha$. By the assumption that $n>2$ it follows that $\alpha\geq 2$. Since $a$ and $b$ are both odd, it follows that
  \begin{equation*}
    \Phi_n(a,b)=a^{2^{\alpha-1}}+b^{2^{\alpha-1}}\equiv 2\pmod{4},
  \end{equation*}
  from which we conclude that $\nu_2(\Phi_n(a,b))=1$.

  For the case where $p$ is odd, write $n=p^\alpha m$ as before. Then we obtain from \zcrefnamed{theorem:lte} that
  \begin{equation*}
    \nu_p(a^n-b^n)-\nu_p(a^m-b^m)=\alpha.
  \end{equation*}
  The right-hand side can be evaluated using the formula $a^n-b^n=\prod_{d\mid n}\Phi_d(a,b)$. Note that for any divisor $d$ of $n$, we have $p\mid\Phi_d(a,b)$ if and only if $d=p^im$ for some $0\leq i\leq \alpha$. Thus we obtain:
  \begin{equation*}
    \sum_{i=1}^\alpha \nu_p(\Phi_{p^im}(a,b))=\alpha.
  \end{equation*}
  Since $p\mid \Phi_{p^im}(a,b)$, this implies that
  \begin{equation*}
    \nu_p(\Phi_{p^im}(a,b))=1
  \end{equation*}
  for every $1\leq i\leq \alpha$. The case $i=\alpha$ proves the claim. 
\end{proof}

\begin{theorem}[Zsigmondy's Theorem]\label{theorem:zsigmondy}
  Consider two integers $1\leq b<a$ with $\gcd(a,b)=1$ and let $n\geq 1$. Then the number $a^n-b^n$ has a primitive prime divisor, except in the following three cases:
  \begin{enumerate}
  \item $n=1$ and $a-b=1$,
  \item $n=2$ and $a+b$ is a power of $2$,
  \item $n=6$, $a=2$, and $b=1$.
  \end{enumerate}
\end{theorem}

\begin{proof}
  In order to dispose of the first case, note that if $n=1$, then any prime divisor of $a-b$ is primitive. Thus, the only case where $a-b$ does not have primitive prime divisors is when $a-b=1$.

  Next, we deal with the case $n=2$. Suppose that $a^2-b^2$ does not have primitive prime divisors. An equivalent way of saying this is that any prime divisor $p\mid \Phi_2(a,b)=a+b$ is also a prime divisor of $a-b$. However, we observe that $\gcd(a+b,a-b)$ divides both $2a$ and $2b$, which implies that the only prime possible prime divisor of $a+b$ is $2$. In other words, $a+b$ is a power of $2$.
  
  For the remainder of the proof we suppose that $n>2$. We will show that if $a^n-b^n$ has no primitive prime divisors, then $n=6$, $a=2$, and $b=1$. By \zcref{proposition:zsigmondy} it follows that if $a^n-b^n$ has no primitive prime divisors, then
  \begin{equation*}
    \Phi_n(a,b)=p,
  \end{equation*}
  where $p$ is the largest prime divisor of $n$ and moreover $p$ is odd. Write $n=p^\alpha m$, where $p\nmid m$. 

  We claim that $m=1$ is impossible. To see this, note that
  \begin{equation*}
    \Phi_{p^\alpha}(a,b)=\frac{a^{p^\alpha}-b^{p^\alpha}}{a^{p^{\alpha-1}}-b^{p^{\alpha-1}}}=a^{p^{\alpha-1}(p-1)}+\cdots + b^{p^{\alpha-1}(p-1)}.
  \end{equation*}
  It follows immediately from this description that $\Phi_{p^\alpha}(a,b)>p$, which contradicts the fact that $\Phi_n(a,b)=p$.

  Next, we claim that $m\geq 3$ is impossible, so that only the case $m=2$ remains. Suppose to the contrary that $m\geq 3$, and define $A:=a^{p^{\alpha-1}}$ and $B:=b^{p^{\alpha-1}}$, so that $\Phi_n(a,b)=\Phi_{pm}(A,B)$. Since $p\nmid m$, we have the identity
  \begin{equation*}
    \Phi_{pm}(A,B) = \frac{\Phi_m(A^p, B^p)}{\Phi_m(A, B)}.
  \end{equation*}
  We estimate the magnitude of the numerator and denominator separately. Recall that $\Phi_m(x,y)=\prod (x-\zeta y)$ where the product runs over the numbers of the form $\zeta=e^{2\pi ik/n}$ where $\gcd(k,m)=1$. Since $m\geq 3$, we note that $\zeta\neq 1$ for every such $k$. Thus, for every such $\zeta$, we have the strict inequalities $|A^p-\zeta B^p| > A^p-B^p$ and $|A-\zeta B| < A+B$. It follows that
  \begin{equation*}
    \Phi_{pm}(A,B) > \frac{(A^p-B^p)^{\totient(m)}}{(A+B)^{\totient(m)}} = \left( \frac{A^p-B^p}{A+B} \right)^{\totient(m)}.
  \end{equation*}
  Since $a>b\geq 1$, we have $A\geq 2$ and $B\geq 1$. Furthermore, $p$ is the largest prime factor of $n$, so $p$ is odd, implying $p\geq 3$. The function $f(x) = (x^p-1)/(x+1)$ is increasing for $x\geq 2$, so the term in the parentheses is at least $(2^p-1)/3$. Combining this with the fact that $\totient(m)\geq 2$ for $m\geq 3$, we obtain
  \begin{equation*}
    \Phi_{pm}(A,B) > \left( \frac{2^p-1}{3} \right)^2.
  \end{equation*}
  For $p=3$, the right-hand side is $(7/3)^2 > 5 > p$. For $p\geq 5$, the inequality holds even more strongly. In all cases we have $\Phi_n(a,b) > p$, which is a contradiction.

  By the previous eliminations, we conclude that $m=2$; that is, $n=2p^\alpha$. It follows that
  \begin{equation*}
    p=\Phi_{2p^\alpha}(a,b)=\frac{A^p+B^p}{A+B}.
  \end{equation*}
  Note that $p(A+B)=A^p+B^p\geq 2(\frac{A+B}{2})^p=\frac{(A+B)^p}{2^{p-1}}$, from which we obtain
  \begin{equation*}
    \left(\frac{3}{2}\right)^{p-1}\leq \left(\frac{A+B}{2}\right)^{p-1}\leq p.
  \end{equation*}
  Since this inequality fails for $p\geq 4$, it follows that $p=3$. We find that
  \begin{equation*}
    \left(\frac{A+B}{2}\right)^{2}\leq 3,
  \end{equation*}
  which implies that $(A+B)^2\leq 12$, and consequently $A+B\leq 3$. It follows that $a^{p^{\alpha-1}}=A=2$ and $B=1$, which gives us that $\alpha=1$, so that $n=6$, $a=2$, and $b=1$.
\end{proof}

\section{The Infinitude of Primes Congruent to \texorpdfstring{$1$}{1} Modulo \texorpdfstring{$n$}{n}}

\begin{theorem}
  For each $n\geq 1$, there are infinitely many primes $p\equiv 1\pmod{n}$. 
\end{theorem}

\begin{proof}
  The statement is self-evident for $n\leq 2$, so we will assume that $n>2$. Consider a finite set $p_1,\ldots,p_k$ of distinct primes such that
  \begin{equation*}
    p_i\equiv 1\pmod{n}
  \end{equation*}
  for each $1\leq i\leq k$, let $a:=n\cdot p_1\cdots p_k$, and define
  \begin{equation*}
    N:=\Phi_n(a).
  \end{equation*}
  Observe that the number $N$ has at least one prime factor. Indeed, this follows directly from \zcrefnamed{theorem:zsigmondy}, since $n,a\geq 3$.

We claim that any prime factor $q\mid N$ satisfies $q\equiv 1\pmod{n}$ and is distinct from the primes $p_i$. Since the constant term of $\Phi_n(x)$ is $1$, it follows that $N\equiv 1\pmod{p_i}$. This immediately implies that $q\neq p_i$ for each $1\leq i\leq k$. We also observe that the congruence $N\equiv 1\pmod{n}$ holds, for the same reason. This shows that any prime factor $q\mid N$ is of the first kind in \zcref{proposition:zsigmondy}, which shows that $q\equiv 1\pmod{n}$.
\end{proof}

\section{Vantieghem's Theorem}

Emmanuel Vantieghem provided a characterization of the prime numbers in terms of the Mersenne numbers \cite{vantieghem1991}. More precisely, he showed that a natural number $n\geq 3$ is prime if and only if
\begin{equation*}
  \prod_{k=1}^{n-1}M_k \equiv n\pmod{M_n}.
\end{equation*}
Recall that the $n$\supth{} Mersenne number is given by $M_n:=2^n-1$. The sequence $(M_n)$ of Mersenne numbers is a strong divisibility sequence. There are some analogies between Vantieghem's theorem and \zcrefnamed{theorem:wilsons-theorem}. In Wilson's theorem, we make use of the congruence
\begin{equation*}
  x^p\equiv x\pmod{p},
\end{equation*}
which holds by \zcrefnamed{theorem:fermats-little-theorem}. For instance, in the second proof of Wilson's theorem, this congruence was used to observe that
\begin{equation*}
  \prod_{k=0}^{p-1}(x-k)\equiv x^p-x\pmod{p}.
\end{equation*}
Similarly, in the proof of Vantieghem's theorem, we will use the congruence $2^p\equiv 1\pmod{M_p}$ to observe that
\begin{equation*}
  \prod_{k=1}^{p-1}(x-2^k)\equiv x^{p}-1\pmod{M_p}.
\end{equation*}
We will prove this congruence in the following proposition.

\begin{proposition}\label{proposition:vantieghem}
  Consider the monic polynomial
  \begin{equation*}
    f(x)=\prod_{k=0}^{p-1}(x-2^k)
  \end{equation*}
  Then we have $f(x)\equiv x^p-1\pmod{M_p}$.
\end{proposition}

\begin{proof}
  To start the proof, we first note that
  \begin{equation*}
    f(0)\equiv\prod_{k=0}^{p-1}(0-2^k)\equiv -2^{\frac{p(p-1)}{2}}\equiv -1\pmod{M_p}.
  \end{equation*}
  This shows that the constant term of $f(x)$ is indeed congruent to $-1$ modulo $M_p$. Since $f(x)$ is a monic polynomial, it therefore suffices to show that its coefficients of degree $0<k<p$ are all $0$.
  
  To see this, we first claim that $f(2x)\equiv f(x)\pmod{M_p}$. Indeed, using that $2^p\equiv 1\pmod{M_p}$, which implies that $2^{-1}\equiv 2^{p-1}\pmod{M_p}$, we see that
  \begin{equation*}
    f(2x)=\prod_{k=0}^{p-1}(2x-2^k)=(2x-1)2^{p-1}\prod_{k=1}^{p-1}(x-2^{k-1})\equiv\prod_{k=0}^{p-1}(x-2^k) \pmod{M_p}.
  \end{equation*}
  Writing $f=a_{p}x^p+\cdots+a_0$, it follows that
  \begin{equation*}
    0\equiv f(2x)-f(x)\equiv \sum_{k=0}^{p}a_k(2^k-1)x^k \pmod{M_p}.
  \end{equation*}
  Since $2^k\equiv 1\pmod{M_p}$ implies $p\mid k$, it follows that $a_k\equiv 0\pmod{M_p}$ for $0<k<p$, which completes the proof.
\end{proof}

Recall from the second proof of \zcrefnamed{theorem:wilsons-theorem}, that after having established that $\prod_k(x-k)\equiv x^p-x\pmod{p}$, we proceeded by taking derivatives on both sides. This strategy again works in the proof of the forward direction of Vantieghem's theorem. For the converse direction, we will make use of \zcrefnamed{theorem:zsigmondy}.

\begin{theorem}[Vantieghem's Theorem]\label{theorem:vantieghem}
  A natural number $n\geq 3$ is prime if and only if
  \begin{equation*}
    \prod_{k=1}^{n-1}M_k \equiv n\pmod{M_n}
  \end{equation*}
\end{theorem}

\begin{proof}
  First, suppose that $n=p$, where $p$ is a prime. Consider the polynomial
  \begin{equation*}
    f(x)=\prod_{k=0}^{p-1}(x-2^k)
  \end{equation*}
  which satisfies the congruence $f(x)\equiv x^p-1\pmod{M_p}$ by \zcref{proposition:vantieghem}. It follows that $f'(1)\equiv p\pmod{M_p}$. On the other hand, writing $f(x)=(x-1)g(x)$, where $g(x)=\prod_{k=1}^{p-1}(x-2^k)$, we find that $f'(x)=(x-1)g'(x)+g(x)$. Evaluating at $1$ gives taht
  \begin{equation*}
    \prod_{k=1}^{p-1}(1-2^k)=g(1)=f'(1)\equiv p\pmod{M_p}.
  \end{equation*}
  Since $p-1$ is even, we can multiply each factor with $-1$ to conclude that $\prod_{k=1}^{p-1}M_k\equiv p\pmod{M_p}$. This completes the proof of the forward direction.

  For the converse, assume that the congruence $\prod_{k=1}^{n-1}M_k \equiv n\pmod{M_n}$ holds. Let $p$ be the largest prime divisor of $n$, so that $M_p\mid M_n$. Then we have the congruence
  \begin{equation*}
    \prod_{k=1}^{n-1}M_k \equiv n\pmod{M_p}
  \end{equation*}
  We will use this congruence to show that if $p$ is a proper divisor of $n$, then there is a prime divisor $q>p$ of $n$. Clearly, such $q$ is again a proper divisor, so there would be by the same argument another prime divisor $r>q$ of $n$, and so on. Since any natural number has only finitely many prime divisors, this shows that $p=n$.
  
  Suppose, as we announced, that $p$ is a proper divisor of $n$. Then $M_p$ is among the factors on the left-hand side, which shows that $0\equiv n\pmod{M_p}$. Thus, we find that $M_p\mid n$. By \zcrefnamed{theorem:zsigmondy} it follows that the Mersenne number $M_p$ has a primitive prime divisor $q\mid M_p$. Indeed, $p$ is an odd prime, so we are not in any of the exceptional cases of Zsigmondy's theorem. Since $q\nmid p$, it follows from \zcref{proposition:zsigmondy} that $q\equiv 1\pmod{p}$. In particular, we must have $q>p$.
\end{proof}

\exercises

\begin{routine}
  \exercise Let $M_n=2^n-1$ be the $n$\supth{} Mersenne number. Show that if $d\mid n$, then
  \begin{equation*}
    \Phi_d(2)\mid M_n.
  \end{equation*}
  \exercise Compute the prime factorization of the Mersenne number $M_{60}=2^{60}-1$.
  \exercise
  \begin{subexenum}
  \item Show that
    \begin{equation*}
      \Phi_n(1)=
      \begin{cases}
        0 & \text{if $n=1$,} \\
        p & \text{if $n=p^k$ for some prime $p$ and $k\geq 1$,} \\
        1 & \text{if two distinct primes divide $n$.}
      \end{cases}
    \end{equation*}
  \item Show that for $n\geq 2$ we have
    \begin{equation*}
      \Phi_n(1) = \frac{\lcm(1,\ldots,n)}{\lcm(1,\ldots,n-1)}.
    \end{equation*}
  \end{subexenum}
  \exercise Let $\sigma(n)$ be the sum of the divisors of a natural number $n$. Show that
  \begin{equation*}
    \sigma(p^m)=\prod_{\substack{d\mid m+1\\d>1}}\Phi_d(p).
  \end{equation*}
  \exercise Suppose that $a$ and $b$ are relatively prime. Show that
  \begin{equation*}
    \frac{2^{ab}-1}{(2^a-1)(2^b-1)}=\prod_{\substack{c\mid a\\ c>1}}\prod_{\substack{d\mid b\\ d>1}} \Phi_{cd}(2).
  \end{equation*}
\end{routine}
