\fp_const:Nn  \c__clock_gear_pressure_angle_fp { 20 * pi / 180 }
\fp_const:Nn  \c__clock_gear_addendum_fp       { 1 }
\fp_const:Nn  \c__clock_gear_dedendum_fp       { 1.25 }
\int_const:Nn \c__clock_gear_flank_steps_int  { 14 }
\int_const:Nn \c__clock_gear_arc_steps_int    { 4 }
\int_const:Nn \c__clock_gear_decimal_places_int { 7 }

\prop_new:N \g__clock_gear_outline_prop
\seq_new:N  \l__clock_gear_tooth_points_seq
\tl_new:N   \l__clock_gear_outline_tl
\int_new:N  \l__clock_gear_teeth_int
\fp_new:N   \l__clock_gear_pitch_radius_fp
\fp_new:N   \l__clock_gear_base_radius_fp
\fp_new:N   \l__clock_gear_tip_radius_fp
\fp_new:N   \l__clock_gear_root_radius_fp
\fp_new:N   \l__clock_gear_flank_start_fp
\fp_new:N   \l__clock_gear_involute_pressure_fp
\fp_new:N   \l__clock_gear_sample_radius_fp
\fp_new:N   \l__clock_gear_local_pressure_fp
\fp_new:N   \l__clock_gear_half_width_fp
\fp_new:N   \l__clock_gear_start_half_width_fp
\fp_new:N   \l__clock_gear_tip_half_width_fp
\fp_new:N   \l__clock_gear_root_gap_fp
\fp_new:N   \l__clock_gear_tooth_angle_fp
\fp_new:N   \l__clock_gear_rotation_cos_fp
\fp_new:N   \l__clock_gear_rotation_sin_fp

\cs_new_protected:Npn \__clock_gear_half_width:n #1
  {
    \fp_set:Nn \l__clock_gear_local_pressure_fp
      { acos( min(1, \l__clock_gear_base_radius_fp / (#1)) ) }
    \fp_set:Nn \l__clock_gear_half_width_fp
      {
        pi / (2 * \l__clock_gear_teeth_int)
        + \l__clock_gear_involute_pressure_fp
        - ( tan(\l__clock_gear_local_pressure_fp)
            - \l__clock_gear_local_pressure_fp )
      }
  }

\cs_new_protected:Npn \__clock_gear_store_polar:nn #1#2
  {
    \seq_put_right:Nx \l__clock_gear_tooth_points_seq
      {
        { \fp_eval:n { (#1) * cos(#2) } }
        { \fp_eval:n { (#1) * sin(#2) } }
      }
  }

\cs_new_protected:Npn \__clock_gear_sample_flank:nn #1#2
  {
    \fp_set:Nn \l__clock_gear_sample_radius_fp
      {
        \l__clock_gear_flank_start_fp
        + (\l__clock_gear_tip_radius_fp - \l__clock_gear_flank_start_fp)
          * (#1) / \c__clock_gear_flank_steps_int
      }
    \__clock_gear_half_width:n { \l__clock_gear_sample_radius_fp }
    \__clock_gear_store_polar:nn
      { \l__clock_gear_sample_radius_fp }
      { (#2) * \l__clock_gear_half_width_fp }
  }

\cs_new_protected:Npn \__clock_gear_build_tooth:
  {
    \seq_clear:N \l__clock_gear_tooth_points_seq
    \__clock_gear_half_width:n { \l__clock_gear_flank_start_fp }
    \fp_set_eq:NN \l__clock_gear_start_half_width_fp
                   \l__clock_gear_half_width_fp
    \__clock_gear_half_width:n { \l__clock_gear_tip_radius_fp }
    \fp_set_eq:NN \l__clock_gear_tip_half_width_fp
                   \l__clock_gear_half_width_fp
    \fp_set:Nn \l__clock_gear_root_gap_fp
      { 2*pi / \l__clock_gear_teeth_int
        - 2*\l__clock_gear_start_half_width_fp }

    \__clock_gear_store_polar:nn
      { \l__clock_gear_root_radius_fp }
      { -\l__clock_gear_start_half_width_fp }
    \int_step_inline:nnnn { 0 } { 1 } { \c__clock_gear_flank_steps_int }
      { \__clock_gear_sample_flank:nn {##1} {-1} }

    \int_step_inline:nn { \c__clock_gear_arc_steps_int }
      {
        \__clock_gear_store_polar:nn
          { \l__clock_gear_tip_radius_fp }
          { -\l__clock_gear_tip_half_width_fp
            + 2*\l__clock_gear_tip_half_width_fp
              * ##1 / \c__clock_gear_arc_steps_int }
      }

    \int_step_inline:nnnn
      { \c__clock_gear_flank_steps_int - 1 } { -1 } { 0 }
      { \__clock_gear_sample_flank:nn {##1} {1} }
    \__clock_gear_store_polar:nn
      { \l__clock_gear_root_radius_fp }
      { \l__clock_gear_start_half_width_fp }

    \int_step_inline:nn { \c__clock_gear_arc_steps_int }
      {
        \__clock_gear_store_polar:nn
          { \l__clock_gear_root_radius_fp }
          { \l__clock_gear_start_half_width_fp
            + \l__clock_gear_root_gap_fp
              * ##1 / \c__clock_gear_arc_steps_int }
      }
  }

\cs_new_protected:Npn \__clock_gear_append_rotated:nn #1#2
  {
    \tl_if_empty:NF \l__clock_gear_outline_tl
      { \tl_put_right:Nn \l__clock_gear_outline_tl { -- } }
    \tl_put_right:Nx \l__clock_gear_outline_tl
      {
        (
          \fp_eval:n
            { round( (#1)*\l__clock_gear_rotation_cos_fp
                   - (#2)*\l__clock_gear_rotation_sin_fp,
                     \c__clock_gear_decimal_places_int ) },
          \fp_eval:n
            { round( (#1)*\l__clock_gear_rotation_sin_fp
                   + (#2)*\l__clock_gear_rotation_cos_fp,
                     \c__clock_gear_decimal_places_int ) }
        )
      }
  }

\cs_new_protected:Npn \__clock_gear_generate:n #1
  {
    \int_set:Nn \l__clock_gear_teeth_int {#1}
    \fp_set:Nn \l__clock_gear_pitch_radius_fp { #1 / 2 }
    \fp_set:Nn \l__clock_gear_base_radius_fp
      { \l__clock_gear_pitch_radius_fp * cos(\c__clock_gear_pressure_angle_fp) }
    \fp_set:Nn \l__clock_gear_tip_radius_fp
      { \l__clock_gear_pitch_radius_fp + \c__clock_gear_addendum_fp }
    \fp_set:Nn \l__clock_gear_root_radius_fp
      { \l__clock_gear_pitch_radius_fp - \c__clock_gear_dedendum_fp }
    \fp_set:Nn \l__clock_gear_flank_start_fp
      { max(\l__clock_gear_base_radius_fp, \l__clock_gear_root_radius_fp) }
    \fp_set:Nn \l__clock_gear_involute_pressure_fp
      { tan(\c__clock_gear_pressure_angle_fp) - \c__clock_gear_pressure_angle_fp }
    \__clock_gear_build_tooth:

    \tl_clear:N \l__clock_gear_outline_tl
    \int_step_inline:nn { \l__clock_gear_teeth_int }
      {
        \fp_set:Nn \l__clock_gear_tooth_angle_fp
          { 2*pi*(##1-1) / \l__clock_gear_teeth_int }
        \fp_set:Nn \l__clock_gear_rotation_cos_fp
          { cos(\l__clock_gear_tooth_angle_fp) }
        \fp_set:Nn \l__clock_gear_rotation_sin_fp
          { sin(\l__clock_gear_tooth_angle_fp) }
        \seq_map_inline:Nn \l__clock_gear_tooth_points_seq
          { \__clock_gear_append_rotated:nn ####1 }
      }
    \tl_put_right:Nn \l__clock_gear_outline_tl { --cycle }
  }
