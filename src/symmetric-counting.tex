\section{Unordered Factorizations}

\begin{theorem}
  A positive natural number $n$ is a square if and only if $\tau(n)$ is odd.
\end{theorem}

\begin{theorem}
  For any positive natural number $n$ we have
  \begin{equation*}
    d_3(n)\equiv
    \begin{cases}
      1 \pmod{3} & \text{if $n$ is a cube,} \\
      0 \pmod{3} & \text{otherwise.}
    \end{cases}
  \end{equation*}
\end{theorem}

\section{Cycle Types of Permutations}

\begin{theorem}
  For each $n$ with partition number $p(n)$, we have $p(n)\mid n!$.
\end{theorem}
