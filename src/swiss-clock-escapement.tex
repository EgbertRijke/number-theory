\newcommand\CEcircleLine[5]{%
 \CEdef{normalAngle}{atan(#3,-(#4))*180/pi}%
 \CEdef{normalLength}{sqrt((#3)^2+(#4)^2)}%
 \CEdef{normalDistance}{(-(#4)*((#1)-\CEex)+(#3)*(#2))/\CEnormalLength}%
 \CEdef{delta}{acosd(max(-1,min(1,\CEnormalDistance/\CElockingRadius)))}%
 \CEnear{candidateA}{\CEnormalAngle+\CEdelta}{#5}%
 \CEnear{candidateB}{\CEnormalAngle-\CEdelta}{#5}%
 \CEdef{heelAngle}{abs(\CEcandidateA-(#5))<abs(\CEcandidateB-(#5))?\CEcandidateA:\CEcandidateB}%
 \CEpolar{contact}{e}{\CElockingRadius}{\CEheelAngle}%
 \CEdef{u}{((\CEcontactx-(#1))*(#3)+(\CEcontacty-(#2))*(#4))/((#3)^2+(#4)^2)}%
}

\newcommand\CEcontact[2]{%
 \CErotate{L}{#2Lock}{p}{#1}\CErotate{J}{#2End}{p}{#1}%
 \edef\CEside{#2}\def\CEentryToken{entry}%
 \ifx\CEside\CEentryToken\CEdef{ref}{180+\CEembrace/2}\else\CEdef{ref}{180-\CEembrace/2}\fi
 \CEdef{faceAngle}{\csname CE#2FaceAngle\endcsname+(#1)}%
 \CEcircleLine{\CELx}{\CELy}{cosd(\CEfaceAngle)}{sind(\CEfaceAngle)}{\CEref}%
 \def\CEstage{locking}%
 \ifnum\fpeval{\CEu < -1e-12?1:0}=1
  \CEcircleLine{\CELx}{\CELy}{\CEJx-\CELx}{\CEJy-\CELy}{\CEref}%
  \def\CEstage{pallet impulse}%
  \ifnum\fpeval{\CEu>1?1:0}=1
   \CEdef{nvx}{-\CEtoothTipy}\CEdef{nvy}{\CEtoothTipx-\CElockingRadius}%
   \CEdef{nl}{sqrt((\CEnvx)^2+(\CEnvy)^2)}%
   \CEdef{nc}{\CEnvx*\CElockingRadius/\CEnl}%
   \CEdef{qangle}{atan(\CEJy,\CEJx-\CEex)*180/pi}%
   \CEdef{nangle}{atan(\CEnvy,\CEnvx)*180/pi}%
   \CEdef{delta}{acosd(\CEnc/sqrt((\CEJx-\CEex)^2+(\CEJy)^2))}%
   \CEnear{candidateA}{\CEqangle-\CEnangle+\CEdelta}{\CEref}%
   \CEnear{candidateB}{\CEqangle-\CEnangle-\CEdelta}{\CEref}%
   \CEdef{heelAngle}{abs(\CEcandidateA-\CEref)<abs(\CEcandidateB-\CEref)?\CEcandidateA:\CEcandidateB}%
   \CEpoint{contact}{\CEJx}{\CEJy}\def\CEstage{tooth impulse}%
  \fi
 \fi
}

\newcommand\CEwheelPose[2]{%
 \edef\CEside{#2}\def\CEentryToken{entry}%
 \ifx\CEside\CEentryToken
  \CEdef{isDrop}{(#1)<-\CEescapingArc/2?1:0}\CEdef{wheelOffset}{0}%
  \def\CEother{exit}\CEdef{landingOffset}{2*360/\CEteeth}%
  \CEdef{releaseAngle}{180+\CEembrace/2-\CEusefulWheelArc}%
 \else
  \CEdef{isDrop}{(#1)>\CEescapingArc/2?1:0}\CEdef{wheelOffset}{2*360/\CEteeth}%
  \def\CEother{entry}\CEdef{landingOffset}{-360/\CEteeth}%
  \CEdef{releaseAngle}{180-\CEembrace/2-\CEusefulWheelArc+\CEwheelOffset}%
 \fi
 \ifnum\CEisDrop=0
  \CEcontact{#1}{#2}\CEdef{wheel}{\CEheelAngle+\CEwheelOffset}%
 \else
  \CEcontact{#1}{\CEother}%
  \CEdef{dropFraction}{min(1,(abs(#1)-\CEescapingArc/2)/\CEdropTravel)}%
  \CEdef{smooth}{(\CEdropFraction)^2*(3-2*\CEdropFraction)}%
  \CEdef{wheel}{\CEreleaseAngle+\CEsmooth*(\CEheelAngle+\CElandingOffset-\CEreleaseAngle)}%
  \ifnum\fpeval{\CEdropFraction<1?1:0}=1\def\CEstage{free drop}\else\def\CEstage{draw to bank}\fi
 \fi
}

\newcommand\ClockEscapementState{%
 \CEdef{beatIndex}{floor(5*\ClockT+.5)}%
 \CEdef{halfTime}{\ClockT-\CEbeatIndex/5}%
 \CEdef{parity}{\CEbeatIndex-2*floor(\CEbeatIndex/2)}%
 \CEdef{progress}{260*sin(5*pi*\CEhalfTime)}%
 \CEdef{balance}{(1-2*\CEparity)*\CEprogress}%
 \CEdef{lever}{abs(\CEprogress)<=\CEhalfEngagement?atan(-\CErollerRadius*sind(\CEbalance),\CEpx-\CErollerRadius*cosd(\CEbalance))*180/pi:-\CEbank*sign(\CEbalance)}%
 \ifnum\CEparity=0\CEwheelPose{\CElever}{entry}\else\CEwheelPose{\CElever}{exit}\fi
 \ifnum\fpeval{abs(\CEprogress)>\CEhalfEngagement?1:0}=1\def\CEstage{locked; balance free}\fi
 \CEdef{unwrappedWheel}{\CEwheel-360/\CEteeth*floor(\CEbeatIndex/2)}%
 \edef\ClockEscapeAngle{\fpeval{\CEunwrappedWheel-360*floor(\CEunwrappedWheel/360)}}%
 \edef\ClockTau{\fpeval{(\CEinitialWheel-\CEunwrappedWheel)/60}}%
 \let\ClockLever\CElever\let\ClockBalance\CEbalance
}
