\section{Definition and Basic Properties of the Gaussian Integers}

\begin{definition}
  A \emph{Gaussian integer} is an expression of the form
  \begin{equation*}
    a+ib
  \end{equation*}
  in which $a$ and $b$ are integers. We say that the \emph{real part} of $a+ib$ is the integer $a$, and the \emph{complex part} of $a+ib$ is the integer $b$.

  We will write $\mathbb{Z}[i]$ for the set of Gaussian integers. Gaussian integers are added, subtracted, and multiplied according to the following rules:
  \begin{align*}
    (a+ib)+(c+id) & = (a+c)+i(b+d) \\
    (a+ib)-(c+id) & = (a-c)+i(b-d) \\
    (a+ib)(c+id)  & = (ac-bd)+i(ad+bc).
  \end{align*}
\end{definition}
