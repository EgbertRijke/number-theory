\chapter{Wieferich Primes}

\begin{definition}
  Let $a$ be a positive integer. A \emph{Wieferich prime base $a$} is a prime number $p$ such that
  \begin{equation*}
    a^{p-1}\equiv 1\pmod{p^2}.
  \end{equation*}
\end{definition}

\begin{enumerate}
\item If $a^{n-1}\equiv 1\pmod{n^2}$ for $n>1$, does that imply that $n$ is prime?

  Suppose $n^2\mid a^{n-1}-1$.
\item
\end{enumerate}

\begin{challenge}
  \exercise Show that if $q^2\mid M_p$ then $q$ is a Wieferich prime. It is an open problem whether every Mersenne number $M_p$ is square-free.
\end{challenge}
